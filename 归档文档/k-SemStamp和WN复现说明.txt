```python
from sklearn.cluster import KMeans
import numpy as np

class KSemStampLogitsProcessor(LogitsProcessor):
    """k-SemStamp: 基于k-means聚类的语义水印"""

    def __init__(self, vocab_size, embedding_model, k=5, gamma=0.5, delta=2.0):
        self.vocab_size = vocab_size
        self.k = k
        self.gamma = gamma
        self.delta = delta

        # 获取词表的语义嵌入
        embeddings = self.get_vocab_embeddings(embedding_model, vocab_size)

        # k-means聚类
        kmeans = KMeans(n_clusters=k, random_state=42)
        self.cluster_labels = kmeans.fit_predict(embeddings)

        # 每个簇的token列表
        self.clusters = [[] for _ in range(k)]
        for token_id, cluster_id in enumerate(self.cluster_labels):
            self.clusters[cluster_id].append(token_id)

    def __call__(self, input_ids, scores):
        # 根据上下文选择一个簇
        context_hash = hash(tuple(input_ids[0].tolist()))
        cluster_id = context_hash % self.k

        # 该簇作为绿名单
        green_list = self.clusters[cluster_id]

        # 偏向绿名单
        for token_id in green_list:
            scores[:, token_id] += self.delta

        return scores
```

```python
def detect_k_semstamp(text, tokenizer, k=5):
    tokens = tokenizer.encode(text)

    # 对每个可能的簇计算绿token比例
    max_score = 0
    for cluster_id in range(k):
        green_count = sum(1 for t in tokens if cluster_labels[t] == cluster_id)
        score = green_count / len(tokens)
        max_score = max(max_score, score)

    # 计算Z-score
    z_score = (max_score - 1/k) / sqrt((1/k) * (1 - 1/k) / len(tokens))
    return z_score
```
